version: "3.8"

services:
  web:
    build:
      context: .
      dockerfile: template-Dockerfile
    container_name: wasa_be
    working_dir: /app
    volumes:
      - .:/app:cached
      - ./media:/app/media
    network_mode: host
    env_file:
      - .env
    environment:
      - REDIS_HOST=127.0.0.1   # Shared host loopback
      - REDIS_PORT=6379
    restart: unless-stopped
    depends_on:
      redis:
        condition: service_healthy

  redis:
    image: redis:7-alpine
    container_name: wasa_redis
    network_mode: host
    # Configures Redis memory limit & automatic LRU eviction for tile caching
    command: redis-server --maxmemory 2gb --maxmemory-policy allkeys-lru --save "" --appendonly no
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5

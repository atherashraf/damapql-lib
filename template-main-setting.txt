def env_bool(name: str, default: bool = False) -> bool:
    """Read a case-insensitive boolean environment variable."""
    value = os.getenv(name)
    if value is None:
        return default
    return value.strip().lower() in {"1", "true", "yes", "on"}


def env_list(name: str, default: str = "") -> list[str]:
    """Read a comma-separated environment variable as a clean list."""
    return [
        item.strip()
        for item in os.getenv(name, default).split(",")
        if item.strip()
    ]


def env_float_list(name: str, default: str) -> list[float]:
    """Read a comma-separated list of numeric values."""
    return [float(value.strip()) for value in os.getenv(name, default).split(",")]

def get_model_bases() -> list[type]:

    during settings and application bootstrap.

        *get_dch_model_bases(),
